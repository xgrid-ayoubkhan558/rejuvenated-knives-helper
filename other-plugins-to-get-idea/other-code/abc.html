document.addEventListener('DOMContentLoaded', function () { const target = document.querySelector('.woocommerce-Price-amount.amount'); if (!target) return; const observer = new MutationObserver(function () { const amountText = target.textContent.trim(); // Extract number only (handles currency symbols & spaces) const numericValue = parseFloat( amountText.replace(/[^\d.,]/g, '').replace(',', '.') ); if (!isNaN(numericValue)) { document.body.setAttribute('data-cart-total', numericValue); console.log('Cart total updated:', numericValue); } }); observer.observe(target, { childList: true, subtree: true, characterData: true }); }); document.body.addEventListener('cart_total_changed', e => { console.log('Received total:', e.detail.total); });
